function getBazaIndexes(headersBaza) {
  // Dodano wyszukiwanie kolumny "wewnetrzna"
  var idx = { ean: 0, nazwa: 1, tempPieczenia: 3, czasSchladzania: 4, tempSchladzania: 5, czas: 6, wewnetrzna: -1 };
  if (!headersBaza) return idx;
  
  for (var i = 0; i < headersBaza.length; i++) {
    var colName = String(headersBaza[i]).toLowerCase().trim();
    if (colName === "ean") idx.ean = i;
    if (colName === "nazwa") idx.nazwa = i;
    if (colName.indexOf("temperatura pieczenia") !== -1) idx.tempPieczenia = i;
    if (colName.indexOf("czas schładzania") !== -1 || colName.indexOf("czas schladzania") !== -1) idx.czasSchladzania = i;
    if (colName.indexOf("temperatura schładzania") !== -1 || colName.indexOf("temperatura schladzania") !== -1) idx.tempSchladzania = i;
    if (colName === "czas") idx.czas = i; 
    if (colName.indexOf("wewnętrzna") !== -1 || colName.indexOf("wewnetrzna") !== -1) idx.wewnetrzna = i; 
  }
  return idx;
}

function calculateExpiryDate(baseDateStr, daysToAdd) {
  if (daysToAdd === "" || daysToAdd === undefined || daysToAdd === null) return "-";
  
  var numStr = String(daysToAdd).replace(/\D/g, '');
  if (numStr === "") return "-";
  
  var numDays = parseInt(numStr, 10);
  var d = new Date(baseDateStr);
  d.setDate(d.getDate() + numDays);
  
  var yyyy = d.getFullYear();
  var mm = ("0" + (d.getMonth() + 1)).slice(-2);
  var dd = ("0" + d.getDate()).slice(-2);
  
  return yyyy + "-" + mm + "-" + dd;
}

function getProducts() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Baza');
  if (!sheet) return [];
  
  var data = sheet.getDataRange().getValues();
  if (data.length < 2) return [];
  var idx = getBazaIndexes(data[0]);
  
  var products = [];
  for (var i = 1; i < data.length; i++) {
    if (data[i][idx.nazwa]) { 
      products.push(data[i][idx.nazwa]);
    }
  }
  return products;
}

  return "Arkusz " + dateStr + " jest gotowy.";
}

function addEntry(dateStr, productName, quantity) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  var bazaSheet = ss.getSheetByName('Baza');
  var bazaData = bazaSheet.getDataRange().getValues();
  var idx = getBazaIndexes(bazaData[0]);
  var productData = null;

  for (var i = 1; i < bazaData.length; i++) {
    if (bazaData[i][idx.nazwa] === productName) {
      productData = bazaData[i];
      break;
    }
  }

  if (!productData) {
    return { success: false, message: "Nie znaleziono produktu w Bazie." };
  }

  var startHour = Math.floor(Math.random() * 9) + 6; 
  var endHour = startHour + 1;
  var tempPieczenia = (productData[idx.tempPieczenia] !== undefined && productData[idx.tempPieczenia] !== "") ? productData[idx.tempPieczenia] : "-";
  var czasSchladzania = (productData[idx.czasSchladzania] !== undefined && productData[idx.czasSchladzania] !== "") ? productData[idx.czasSchladzania] : "-";
  var tempSchladzania = (productData[idx.tempSchladzania] !== undefined && productData[idx.tempSchladzania] !== "") ? productData[idx.tempSchladzania] : "-";
  var tempWewnetrzna = (idx.wewnetrzna !== -1 && productData[idx.wewnetrzna] !== undefined && productData[idx.wewnetrzna] !== "") ? productData[idx.wewnetrzna] : "-";
  
  var waznoscStr = calculateExpiryDate(dateStr, productData[idx.czas]);

  // Tablica powiększona o 11 element
  var rowData = [
    productName, startHour, endHour, tempPieczenia, tempWewnetrzna, quantity, 
    endHour, czasSchladzania, tempSchladzania, waznoscStr, "-"
  ];

  var sheetIndex = 1;
  var sheetName = dateStr;
  var currentSheet = ss.getSheetByName(sheetName);
  var emptyRowIndex = -1;
  var maxDataRows = 27;

  while (true) {
    if (!currentSheet) {
      currentSheet = createSheetIfNotExist(sheetName, dateStr);
    }
    var dataRange = currentSheet.getRange(8, 1, maxDataRows, 1).getValues();
    emptyRowIndex = -1;
    for (var j = 0; j < dataRange.length; j++) {
      if (dataRange[j][0] === "") {
        emptyRowIndex = 8 + j;
        break;
      }
    }
    if (emptyRowIndex !== -1) break; 
    sheetIndex++;
    sheetName = dateStr + " nr " + sheetIndex;
    currentSheet = ss.getSheetByName(sheetName);
  }

  currentSheet.getRange(emptyRowIndex, 1, 1, rowData.length).setValues([rowData]);
  ss.setActiveSheet(currentSheet);
  return { success: true, message: "Pomyślnie dodano: " + productName };
}

function importFromFormularz(dateStr) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  var formSheet = ss.getSheetByName('Formularz');
  if (!formSheet) return { success: false, message: "Brak arkusza o nazwie 'Formularz'!" };
  
  var bazaSheet = ss.getSheetByName('Baza');
  if (!bazaSheet) return { success: false, message: "Brak arkusza o nazwie 'Baza'!" };
  
  var bazaData = bazaSheet.getDataRange().getValues();
  var formData = formSheet.getDataRange().getValues();
  var idx = getBazaIndexes(bazaData[0]);
  
  var mapEan = {};
  var mapNazwa = {};
  for (var i = 1; i < bazaData.length; i++) {
    var ean = String(bazaData[i][idx.ean]).trim();
    var nazwa = String(bazaData[i][idx.nazwa]).trim();
    if (ean) mapEan[ean] = bazaData[i];
    if (nazwa) mapNazwa[nazwa.toLowerCase()] = bazaData[i];
  }
  
  var entriesToAdd = [];
  var notFoundCount = 0;
  
  for (var r = 1; r < formData.length; r++) {
    var row = formData[r];
    var eanForm = String(row[1] || "").trim();               
    var nazwaForm = String(row[2] || "").trim().toLowerCase(); 
    var qtyForm = parseFloat(row[5]);                        
    
    if (isNaN(qtyForm) || qtyForm <= 0) qtyForm = 1; 
    if (!eanForm && !nazwaForm) continue; 
    
    var productData = null;
    
    if (eanForm && mapEan[eanForm]) {
      productData = mapEan[eanForm];
    } else if (nazwaForm && mapNazwa[nazwaForm]) {
      productData = mapNazwa[nazwaForm];
    }
    
    if (productData) {
      var startHour = Math.floor(Math.random() * 9) + 6; 
      var endHour = startHour + 1;
      var tempPieczenia = (productData[idx.tempPieczenia] !== undefined && productData[idx.tempPieczenia] !== "") ? productData[idx.tempPieczenia] : "-";
      var czasSchladzania = (productData[idx.czasSchladzania] !== undefined && productData[idx.czasSchladzania] !== "") ? productData[idx.czasSchladzania] : "-";
      var tempSchladzania = (productData[idx.tempSchladzania] !== undefined && productData[idx.tempSchladzania] !== "") ? productData[idx.tempSchladzania] : "-";
      var tempWewnetrzna = (idx.wewnetrzna !== -1 && productData[idx.wewnetrzna] !== undefined && productData[idx.wewnetrzna] !== "") ? productData[idx.wewnetrzna] : "-";
      
      var waznoscStr = calculateExpiryDate(dateStr, productData[idx.czas]);

      entriesToAdd.push([
        productData[idx.nazwa], 
        startHour, endHour, tempPieczenia, tempWewnetrzna, qtyForm, 
        endHour, czasSchladzania, tempSchladzania, waznoscStr, "-"
      ]);
    } else {
      notFoundCount++;
    }
  }
  
  if (entriesToAdd.length === 0) {
    return { success: false, message: "Nie znaleziono asortymentu do dodania (zły kod EAN lub Nazwa)." };
  }
  
  var maxDataRows = 27;
  var currentSheetIndex = 1;
  var sheetName = dateStr;
  var currentSheet = ss.getSheetByName(sheetName);
  var addedCount = 0;
  
  for (var k = 0; k < entriesToAdd.length; k++) {
    var rowData = entriesToAdd[k];
    var emptyRowIndex = -1;
    
    while (true) {
      if (!currentSheet) currentSheet = createSheetIfNotExist(sheetName, dateStr);
      var dataRange = currentSheet.getRange(8, 1, maxDataRows, 1).getValues();
      emptyRowIndex = -1;
      
      for (var j = 0; j < dataRange.length; j++) {
        if (dataRange[j][0] === "") {
          emptyRowIndex = 8 + j;
          break;
        }
      }
      if (emptyRowIndex !== -1) break;
      currentSheetIndex++;
      sheetName = dateStr + " nr " + currentSheetIndex;
      currentSheet = ss.getSheetByName(sheetName);
    }
    
    currentSheet.getRange(emptyRowIndex, 1, 1, rowData.length).setValues([rowData]);
    addedCount++;
  }
  
  ss.setActiveSheet(currentSheet);
  
  var finalMsg = "Zakończono. Pomyślnie dodano wpisów: " + addedCount;
  if (notFoundCount > 0) finalMsg += "\n(Brak asortymentu w Bazie: " + notFoundCount + ")";
  
  return { success: true, message: finalMsg };
}

function undoLastEntry(dateStr) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var maxDataRows = 27;
  
  var maxIdx = 1;
  while(ss.getSheetByName(dateStr + " nr " + (maxIdx + 1))) {
    maxIdx++;
  }
  
  for (var i = maxIdx; i >= 1; i--) {
    var sheetName = (i === 1) ? dateStr : dateStr + " nr " + i;
    var sheet = ss.getSheetByName(sheetName);
    if(!sheet) continue;
    
    var dataRange = sheet.getRange(8, 1, maxDataRows, 1).getValues();
    
    for (var r = maxDataRows - 1; r >= 0; r--) {
      if (dataRange[r][0] !== "") {
        // Czyszczenie 11 kolumn
        var emptyRow = [["", "", "", "", "", "", "", "", "", "", ""]];
        sheet.getRange(8 + r, 1, 1, 11).setValues(emptyRow);
        return { success: true, message: "Cofnięto najświeższy wpis z arkusza " + sheetName };
      }
    }
  }
  return { success: false, message: "Brak wpisów do cofnięcia w wybranym dniu." };
}
